<%*
/* サムネイルを選ぶ（Templater）  ショートカット: Alt + T
   一覧に表示するサムネイル画像と、正方形に切り抜く位置を選ぶ。
   設定は日本語ページ（index.md）に保存され、英語版も同じサムネイルを使う。
   何も指定しなければ「本文の1枚目の画像」がサムネイルになる。 */
const IMG = /\.(jpe?g|png|gif|webp|svg)$/i;
const f = tp.config.target_file;

if (!/^index(\.en)?\.md$/.test(f.name)) {
  new Notice("ページ（index.md または index.en.md）を開いた状態で実行してください。", 8000);
} else {
  const ja = app.vault.getAbstractFileByPath(f.parent.path + "/index.md") || f;
  const fm = (app.metadataCache.getFileCache(ja) || {}).frontmatter || {};
  const imgs = f.parent.children
    .filter(x => x.extension && IMG.test(x.name))
    .sort((a, b) => a.name.localeCompare(b.name));

  const labels = ["（指定なし）本文の1枚目の画像を使う"].concat(
    imgs.map(x => x.name + (fm.thumbnail === x.name ? "　← いまのサムネイル" : "")));
  const values = [""].concat(imgs.map(x => x.name));
  const pick = await tp.system.suggester(labels, values, false, "サムネイルにする画像を選んでください");

  if (pick !== null && pick !== undefined) {
    const posLabels = ["中央（標準）", "上を見せる", "下を見せる", "左を見せる", "右を見せる"];
    const posValues = ["", "top", "bottom", "left", "right"];
    const pos = await tp.system.suggester(posLabels, posValues, false, "正方形に切り抜くとき、どこを見せますか");

    await app.fileManager.processFrontMatter(ja, d => {
      if (pick) d.thumbnail = pick; else delete d.thumbnail;
      if (pos) d.thumbnail_position = pos; else delete d.thumbnail_position;
    });
    new Notice("サムネイルを設定しました: " + (pick || "本文の1枚目") + (pos ? "（" + posLabels[posValues.indexOf(pos)] + "）" : "") + "\n英語版も同じサムネイルになります。", 6000);
  }
}
-%>
