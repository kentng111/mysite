<%*
/* 画像を挿入（Templater）  ショートカット: Alt + I
   1) このページのフォルダにある画像を一覧から選ぶ
   2) 名前が半角英数字でなければ、整えた名前に変更（リンクも自動で更新）
   3) カーソルの位置に挿入
   4) もう一方の言語（index.md ⇔ index.en.md）にも、同じ順番の位置に挿入 */
const IMG = /\.(jpe?g|png|gif|webp|svg)$/i;
const isImgLine = l => /^\s*!\[[^\]]*\]\([^)\s]+/.test(l);
const f = tp.config.target_file;
const editor = app.workspace.activeEditor && app.workspace.activeEditor.editor;

if (!/^index(\.en)?\.md$/.test(f.name) || !editor) {
  new Notice("ページ（index.md または index.en.md）を開いた状態で実行してください。", 8000);
} else {
  const text = editor.getValue();
  const used = new Set((text.match(/!\[[^\]]*\]\([^)\s]+/g) || []).map(m => decodeURI(m.replace(/^.*\]\(/, ""))));
  const imgs = f.parent.children
    .filter(x => x.extension && IMG.test(x.name))
    .sort((a, b) => a.name.localeCompare(b.name));

  if (!imgs.length) {
    new Notice("このページのフォルダに画像がありません。\n左のファイル一覧のこのページのフォルダに、画像をドラッグして入れてから実行してください。", 10000);
  } else {
    const labels = imgs.map(x =>
      x.name + (x.name.startsWith("cover_") ? "　［サムネイル］" : "") + (used.has(x.name) ? "　（使用済み）" : ""));
    const img = await tp.system.suggester(labels, imgs, false, "挿入する画像を選んでください");

    if (img) {
      // --- 名前を整える（半角小文字・数字・ハイフン・アンダーバー、拡張子は小文字） ---
      const ext = img.extension.toLowerCase().replace("jpeg", "jpg");
      if (!/^[a-z0-9_-]+$/.test(img.basename) || img.extension !== ext) {
        let sug = img.basename.toLowerCase().replace(/\s+/g, "-").replace(/[^a-z0-9_-]/g, "").replace(/-+/g, "-").replace(/^-|-$/g, "");
        if (!sug || sug === "pasted-image") sug = "img-" + tp.date.now("YYYYMMDD-HHmmss");
        const name = await tp.system.prompt("画像の名前を半角英数字に整えます（拡張子は不要。サムネイルは cover_ で始める）", sug);
        const clean = (name || "").toLowerCase().replace(/[^a-z0-9_-]+/g, "-").replace(/^-|-$/g, "");
        if (clean) {
          const newPath = f.parent.path + "/" + clean + "." + ext;
          if (newPath !== img.path && app.vault.getAbstractFileByPath(newPath)) {
            new Notice("同じ名前の画像があるため、名前は変えずに挿入します: " + clean + "." + ext, 8000);
          } else if (newPath !== img.path) {
            const oldName = img.name;
            await app.fileManager.renameFile(img, newPath);
            // サムネイルに指定されていた場合は、指定も新しい名前にする
            for (const n of ["index.md", "index.en.md"]) {
              const pf = app.vault.getAbstractFileByPath(f.parent.path + "/" + n);
              if (pf) await app.fileManager.processFrontMatter(pf, d => { if (d.thumbnail === oldName) d.thumbnail = img.name; });
            }
          }
        }
      }
      const md = "![](" + img.name + ")";

      // --- このページに挿入（空の行ならその行に、文字がある行ならその下に） ---
      const cur = editor.getCursor();
      const lineText = editor.getLine(cur.line);
      const k = editor.getValue().split(/\r?\n/).slice(0, cur.line + 1).filter(isImgLine).length;
      if (lineText.trim() === "") {
        editor.replaceRange(md, { line: cur.line, ch: 0 }, { line: cur.line, ch: lineText.length });
      } else {
        editor.replaceRange("\n\n" + md, { line: cur.line, ch: lineText.length });
      }

      // --- もう一方の言語にも、同じ順番の位置に挿入 ---
      const otherPath = f.parent.path + "/" + (f.name === "index.md" ? "index.en.md" : "index.md");
      const other = app.vault.getAbstractFileByPath(otherPath);
      if (other) {
        await app.vault.process(other, data => {
          const nl = data.includes("\r\n") ? "\r\n" : "\n";
          let L = data.split(/\r?\n/);
          let start = 0;
          if (L[0] === "---") { const e = L.indexOf("---", 1); if (e > 0) start = e + 1; }
          const imgIdx = [];
          for (let i = start; i < L.length; i++) if (isImgLine(L[i])) imgIdx.push(i);
          if (k < imgIdx.length) {
            const at = imgIdx[k];
            const ins = (at > start && L[at - 1].trim() !== "") ? ["", md, ""] : [md, ""];
            L.splice(at, 0, ...ins);
          } else {
            while (L.length > start && L[L.length - 1].trim() === "") L.pop();
            if (L.length > start) L.push("");
            L.push(md, "");
          }
          return L.join(nl);
        });
        new Notice((f.name === "index.md" ? "英語版" : "日本語版") + "にも同じ順番で挿入しました。文章の中での位置は、必要に応じて調整してください。", 6000);
      }
    }
  }
}
-%>
