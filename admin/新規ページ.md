<%*
/* 新規ページを作成（Templater）
   使い方: コマンドパレット →「Templater: Create new note from template」→ このテンプレート
           または Alt+N
   1) タイトルとURL名を聞く  2) content/<URL名>/index.md を作る  3) 下のひな形を入れる */
let ok = false, title = "", slug = "";
const current = tp.config.target_file;
const isEmpty = (await app.vault.read(current)).trim() === "";
if (!isEmpty) {
  new Notice("既存のページでは実行できません。新しい空のノートで実行してください。", 8000);
} else {
  title = (await tp.system.prompt("ページのタイトル（日本語でOK）")) || "";
  if (title) {
    const raw = await tp.system.prompt("URL名（半角英数字とハイフン。例: tenjin-house）", "");
    slug = (raw || "").toLowerCase().trim().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "");
  }
  const dir = "content/" + slug;
  if (!title || !slug) {
    new Notice("タイトルかURL名が空のため中止しました。空のノートは削除してください。", 8000);
  } else if (app.vault.getAbstractFileByPath(dir)) {
    new Notice("同じURL名のページが既にあります: " + slug, 8000);
  } else {
    await app.vault.createFolder(dir);
    await tp.file.move(dir + "/index");
    ok = true;
  }
}
const q = s => '"' + s.replace(/\\/g, "\\\\").replace(/"/g, '\\"') + '"';
-%>
<%* if (ok) { -%>
---
title: <% q(title) %>
slug: <% slug %>
description: 
sort_title: 
tags: 
draft: true
color: "#2C5159"
year: <% tp.date.now("YYYY-MM") %>
index: a999
neko: true
thumbnail: 
thumbnail_position: 
---

<!-- 画像はこのフォルダに入れて Alt + I で挿入。サムネイルは本文の1枚目（変えるときは Alt + T） -->

<%* } -%>
