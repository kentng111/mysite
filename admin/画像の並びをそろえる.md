<%*
/* 画像の並びをそろえる（Templater）  ショートカット: Alt + S
   いま開いているページの画像の並びを正として、もう一方の言語のページの画像を同じ並びに置き換える。
   もう一方のページの文章はそのまま残す（画像と画像の間の文章は、同じ区切りの位置に残る） */
const isImgLine = l => /^\s*!\[[^\]]*\]\([^)\s]+/.test(l);
const f = tp.config.target_file;
const editor = app.workspace.activeEditor && app.workspace.activeEditor.editor;

if (!/^index(\.en)?\.md$/.test(f.name) || !editor) {
  new Notice("ページ（index.md または index.en.md）を開いた状態で実行してください。", 8000);
} else {
  const isJa = f.name === "index.md";
  const otherName = isJa ? "英語版" : "日本語版";
  const other = app.vault.getAbstractFileByPath(f.parent.path + "/" + (isJa ? "index.en.md" : "index.md"));
  if (!other) {
    new Notice(otherName + "がありません。" + (isJa ? "Alt + E で英語版を作成できます。" : ""), 8000);
  } else {
    const master = editor.getValue().split(/\r?\n/).filter(isImgLine).map(l => l.trim());
    const ok = await tp.system.suggester(
      ["そろえる（" + otherName + "の画像を、このページと同じ " + master.length + " 枚・同じ並びにする）", "やめる"],
      [true, false], false, otherName + "の画像の並びを、このページに合わせます");
    if (ok) {
      await app.vault.process(other, data => {
        const nl = data.includes("\r\n") ? "\r\n" : "\n";
        const L = data.split(/\r?\n/);
        let start = 0;
        if (L[0] === "---") { const e = L.indexOf("---", 1); if (e > 0) start = e + 1; }
        // 本文を「文章のかたまり」と「画像」に分ける
        const segs = [[]];
        for (let i = start; i < L.length; i++) {
          if (isImgLine(L[i])) segs.push([]); else segs[segs.length - 1].push(L[i]);
        }
        const out = L.slice(0, start).concat(segs[0]);
        const pushBlank = () => { if (out.length > start && out[out.length - 1].trim() !== "") out.push(""); };
        master.forEach((md, i) => {
          pushBlank();
          out.push(md);
          if (segs[i + 1]) out.push(...segs[i + 1]); else out.push("");
        });
        // 画像が減った場合、余った文章は最後に残す
        for (let j = master.length + 1; j < segs.length; j++) { pushBlank(); out.push(...segs[j]); }
        while (out.length > start + 1 && out[out.length - 1].trim() === "" && out[out.length - 2].trim() === "") out.pop();
        return out.join(nl);
      });
      new Notice(otherName + "の画像の並びを、このページに合わせました（" + master.length + " 枚）。", 6000);
    }
  }
}
