<%*
/* 英語版を作成（Templater）
   使い方: 日本語ページ（index.md）を開いた状態で Alt+E
   同じフォルダに index.en.md を作り、共通の項目と画像の行をコピーして開きます */
const f = tp.config.target_file;
const enPath = f.parent.path + "/index.en.md";
if (f.name !== "index.md") {
  new Notice("日本語ページ（index.md）を開いた状態で実行してください。", 8000);
} else if (app.vault.getAbstractFileByPath(enPath)) {
  new Notice("英語版は既にあります。開きます。");
  await app.workspace.openLinkText(enPath, "", true);
} else {
  const fm = (app.metadataCache.getFileCache(f) || {}).frontmatter || {};
  const title = (await tp.system.prompt("英語のタイトル", fm.title || "")) || fm.title || "";
  const q = s => '"' + String(s).replace(/\\/g, "\\\\").replace(/"/g, '\\"') + '"';
  const tags = Array.isArray(fm.tags) ? fm.tags : (fm.tags ? [fm.tags] : []);
  const body = await app.vault.read(f);
  const images = body.split(/\r?\n/).filter(l => /^\s*!\[/.test(l));
  const lines = [
    "---",
    "title: " + q(title),
    "slug: " + (fm.slug || ""),
    "description: ",
    "sort_title: ",
    "tags:" + (tags.length ? "" : " "),
    ...tags.map(t => "  - " + t),
    "draft: " + (fm.draft === false ? "false" : "true"),
    "color: " + q(fm.color || "#2C5159"),
    "year: " + (fm.year || ""),
    "index: " + (fm.index || ""),
    "neko: " + (fm.neko === false ? "false" : "true"),
    "---",
    "",
    "<!-- English text here. Images are shared with the Japanese page. -->",
    "",
    ...images.flatMap(l => [l, ""])
  ];
  await app.vault.create(enPath, lines.join("\n"));
  await app.workspace.openLinkText(enPath, "", true);
}
-%>
